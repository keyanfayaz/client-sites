```typescript
// API Route: User authentication with JWT
import { Request, Response } from 'express';
import jwt from 'jsonwebtoken';
import bcrypt from 'bcrypt';
import { UserService } from '../services/UserService';

interface AuthRequest extends Request {
  body: {
    email: string;
    password: string;
  };
}

export async function loginHandler(req: AuthRequest, res: Response) {
  try {
    const { email, password } = req.body;
    
    // Validate user credentials
    const user = await UserService.findByEmail(email);
    if (!user || !await bcrypt.compare(password, user.passwordHash)) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }

    // Generate JWT token
    const token = jwt.sign(
      { userId: user.id, email: user.email },
      process.env.JWT_SECRET!,
      { expiresIn: '24h' }
    );

    res.json({
      user: { id: user.id, email: user.email, name: user.name },
      token,
      expiresIn: '24h'
    });
  } catch (error) {
    console.error('Login error:', error);
    res.status(500).json({ error: 'Internal server error' });
  }
}
```

```python
# API: Advanced data processing with async/await
from fastapi import FastAPI, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import List, Optional
import asyncio
import pandas as pd

app = FastAPI(title="Beta Analytics API", version="2.0.0")

class DataProcessingRequest(BaseModel):
    dataset_id: str
    operations: List[str]
    output_format: str = "json"

class ProcessingResult(BaseModel):
    job_id: str
    status: str
    result_url: Optional[str] = None

@app.post("/api/v2/process-data", response_model=ProcessingResult)
async def process_data(
    request: DataProcessingRequest, 
    background_tasks: BackgroundTasks
):
    """
    Process large datasets asynchronously with real-time progress tracking
    """
    job_id = f"job_{request.dataset_id}_{int(time.time())}"
    
    # Start background processing
    background_tasks.add_task(
        process_dataset_async, 
        job_id, 
        request.dataset_id, 
        request.operations
    )
    
    return ProcessingResult(
        job_id=job_id,
        status="processing"
    )

async def process_dataset_async(
    job_id: str, 
    dataset_id: str, 
    operations: List[str]
):
    """Background task for heavy data processing"""
    try:
        # Load dataset
        df = await load_dataset(dataset_id)
        
        # Apply operations
        for operation in operations:
            df = await apply_operation(df, operation)
            
        # Save results
        result_url = await save_results(job_id, df)
        
        # Update job status
        await update_job_status(job_id, "completed", result_url)
        
    except Exception as e:
        await update_job_status(job_id, "failed", error=str(e))
```

```sql
-- Complex analytics query with window functions
WITH monthly_metrics AS (
  SELECT 
    DATE_TRUNC('month', created_at) as month,
    user_id,
    COUNT(*) as actions_count,
    SUM(revenue) as monthly_revenue,
    LAG(SUM(revenue)) OVER (
      PARTITION BY user_id 
      ORDER BY DATE_TRUNC('month', created_at)
    ) as previous_month_revenue
  FROM user_actions 
  WHERE created_at >= NOW() - INTERVAL '12 months'
  GROUP BY DATE_TRUNC('month', created_at), user_id
),
growth_metrics AS (
  SELECT *,
    CASE 
      WHEN previous_month_revenue IS NOT NULL 
      THEN ((monthly_revenue - previous_month_revenue) / previous_month_revenue) * 100
      ELSE NULL 
    END as growth_rate
  FROM monthly_metrics
)
SELECT 
  month,
  COUNT(DISTINCT user_id) as active_users,
  AVG(monthly_revenue) as avg_revenue_per_user,
  AVG(growth_rate) as avg_growth_rate,
  PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY monthly_revenue) as median_revenue
FROM growth_metrics
GROUP BY month
ORDER BY month DESC;
```

### Redis for Caching & Real-time Features
```javascript
// High-performance caching with Redis
import Redis from 'ioredis';

class CacheService {
  constructor() {
    this.redis = new Redis({
      host: process.env.REDIS_HOST,
      port: process.env.REDIS_PORT,
      retryDelayOnFailover: 100,
      maxRetriesPerRequest: 3
    });
  }

  // Cache with automatic expiration
  async setWithTTL(key, value, ttlSeconds = 3600) {
    await this.redis.setex(key, ttlSeconds, JSON.stringify(value));
  }

  // Get with fallback to database
  async getOrFetch(key, fetchFunction, ttlSeconds = 3600) {
    const cached = await this.redis.get(key);
    
    if (cached) {
      return JSON.parse(cached);
    }

    const fresh = await fetchFunction();
    await this.setWithTTL(key, fresh, ttlSeconds);
    return fresh;
  }

  // Real-time pub/sub for live updates
  async publishUpdate(channel, data) {
    await this.redis.publish(channel, JSON.stringify(data));
  }

  subscribeToUpdates(channel, callback) {
    const subscriber = this.redis.duplicate();
    subscriber.subscribe(channel);
    subscriber.on('message', (channel, message) => {
      callback(JSON.parse(message));
    });
    return subscriber;
  }
}
```

```typescript
// Example: Efficient data loading with caching and pagination
import { LRUCache } from 'lru-cache';

class DataService {
  private cache = new LRUCache<string, any>({
    max: 1000,
    ttl: 1000 * 60 * 10 // 10 minutes
  });

  async getPagedData(page: number, limit: number, filters: any = {}) {
    const cacheKey = `data:${page}:${limit}:${JSON.stringify(filters)}`;
    
    // Check cache first
    const cached = this.cache.get(cacheKey);
    if (cached) return cached;
    
    // Database query with optimized indexing
    const offset = (page - 1) * limit;
    const query = `
      SELECT 
        id, name, email, created_at,
        COUNT(*) OVER() as total_count
      FROM users 
      WHERE ($1::text IS NULL OR name ILIKE $1)
      ORDER BY created_at DESC
      LIMIT $2 OFFSET $3
    `;
    
    const searchTerm = filters.search ? `%${filters.search}%` : null;
    const result = await db.query(query, [searchTerm, limit, offset]);
    
    const response = {
      data: result.rows,
      pagination: {
        page,
        limit,
        total: result.rows[0]?.total_count || 0,
        hasMore: result.rows.length === limit
      }
    };
    
    // Cache the result
    this.cache.set(cacheKey, response);
    return response;
  }
}
```
